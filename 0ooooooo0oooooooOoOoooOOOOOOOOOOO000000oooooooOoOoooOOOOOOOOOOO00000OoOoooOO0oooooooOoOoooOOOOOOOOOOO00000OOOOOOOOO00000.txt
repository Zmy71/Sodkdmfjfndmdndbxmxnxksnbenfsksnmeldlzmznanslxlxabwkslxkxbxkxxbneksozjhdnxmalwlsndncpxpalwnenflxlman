local function createMainUI()

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0,270,0,175)
    frame.Position = UDim2.new(.5,-135,.5,-87)
    frame.BackgroundColor3 = Color3.fromRGB(5,5,5)
    frame.Parent = gui
    frame.Active = true
    frame.Draggable = true
    Instance.new("UICorner", frame)

    local stroke2 = Instance.new("UIStroke")
    stroke2.Thickness = 2
    stroke2.Color = Color3.fromRGB(0,255,0)
    stroke2.Parent = frame

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1,-30,0,30)
    title.BackgroundTransparency = 1
    title.Text = "🏁 AUTO SYSTEM"
    title.Font = Enum.Font.GothamBlack
    title.TextColor3 = Color3.fromRGB(0,255,255)
    title.TextScaled = true
    title.Parent = frame

    local closeBtn = Instance.new("TextButton")
    closeBtn.Size = UDim2.new(0,30,0,30)
    closeBtn.Position = UDim2.new(1,-30,0,0)
    closeBtn.Text = "X"
    closeBtn.BackgroundColor3 = Color3.fromRGB(255,0,0)
    closeBtn.TextColor3 = Color3.new(1,1,1)
    closeBtn.Parent = frame
    Instance.new("UICorner", closeBtn)

    closeBtn.MouseButton1Click:Connect(function()
        gui:Destroy()
    end)

    local info = Instance.new("TextLabel")
    info.Size = UDim2.new(1,-10,0,60)
    info.Position = UDim2.new(0,5,0,32)
    info.BackgroundTransparency = 1
    info.TextWrapped = true
    info.Font = Enum.Font.Code
    info.TextSize = 13
    info.TextColor3 = Color3.fromRGB(0,255,255)
    info.Text = "READY"
    info.Parent = frame

    local scanBtn = Instance.new("TextButton")
    scanBtn.Size = UDim2.new(.4,0,0,35)
    scanBtn.Position = UDim2.new(.05,0,1,-45)
    scanBtn.Text = "SCAN"
    scanBtn.BackgroundColor3 = Color3.fromRGB(0,255,255)
    scanBtn.TextColor3 = Color3.new(0,0,0)
    scanBtn.Parent = frame
    Instance.new("UICorner", scanBtn)

    local tpBtn = Instance.new("TextButton")
    tpBtn.Size = UDim2.new(.4,0,0,35)
    tpBtn.Position = UDim2.new(.55,0,1,-45)
    tpBtn.Text = "AUTO TP"
    tpBtn.BackgroundColor3 = Color3.fromRGB(255,0,255)
    tpBtn.TextColor3 = Color3.new(0,0,0)
    tpBtn.Parent = frame
    Instance.new("UICorner", tpBtn)

    local function isCheckpoint(v)
        return v:IsA("Part") and v.Name == "Point" and v.Parent and tonumber(v.Parent.Name)
    end

    local function scanAll()
        local found = {}

        local function recurse(parent)
            for _, v in ipairs(parent:GetChildren()) do
                if isCheckpoint(v) then
                    table.insert(found, v)
                end
                recurse(v)
            end
        end

        recurse(game.Workspace)
        return found
    end

    local checkpoints = {}
    local running = false

    scanBtn.MouseButton1Click:Connect(function()
        checkpoints = scanAll()
        info.Text = "FOUND: "..#checkpoints
    end)

    tpBtn.MouseButton1Click:Connect(function()
        running = not running
        if not running then return end

        task.spawn(function()
            for i, cp in ipairs(checkpoints) do
                if not running then break end

                local char = player.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")

                if cp and root then
                    root.CFrame = CFrame.new(cp.Position + Vector3.new(0,3,0))
                    info.Text = "STAGE "..cp.Parent.Name
                    task.wait(0.5)
                end
            end

            running = false
            info.Text = "DONE"
        end)
    end)
end
