        game.CoreGui.AutoCheckpointObby:Destroy()
    end)

    Instance.new("UICorner", frame)

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1,-30,0,30)
    title.BackgroundTransparency = 1
    title.Text = "🏁 Auto Checkpoint Obby"
    title.Font = Enum.Font.GothamBold
    title.TextColor3 = Color3.new(1,1,1)
    title.TextScaled = true
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Parent = frame

    local closeBtn = Instance.new("TextButton")
    closeBtn.Size = UDim2.new(0,30,0,30)
    closeBtn.Position = UDim2.new(1,-30,0,0)
    closeBtn.Text = "X"

    closeBtn.TextColor3 = Color3.new(1,1,1)
    closeBtn.BackgroundColor3 = Color3.fromRGB(200,50,50)
    closeBtn.Parent = frame
    Instance.new("UICorner", closeBtn)

    closeBtn.MouseButton1Click:Connect(function()
        gui:Destroy()
    end)

    local toggleBtn = Instance.new("TextButton")
    toggleBtn.Size = UDim2.new(0,30,0,30)
    toggleBtn.Position = UDim2.new(0,5,0,5)
    toggleBtn.Text = "≡"
    toggleBtn.Parent = gui

    Instance.new("UICorner", toggleBtn)

    local visible = true
    toggleBtn.MouseButton1Click:Connect(function()
        visible = not visible
        frame.Visible = visible
    end)

    local info = Instance.new("TextLabel")
    info.Size = UDim2.new(1,-10,0,60)
    info.Position = UDim2.new(0,5,0,32)
    info.BackgroundTransparency = 1
    info.TextWrapped = true
    info.Font = Enum.Font.Code
    info.TextSize = 13
    info.TextColor3 = Color3.fromRGB(200,200,200)
    info.Text = "Tekan Scan dulu, lalu Auto TP"
    info.Parent = frame

    local scanBtn = Instance.new("TextButton")
    scanBtn.Size = UDim2.new(.4,0,0,35)
    scanBtn.Position = UDim2.new(.05,0,1,-45)
    scanBtn.Text = "🔍 Scan"
    scanBtn.Parent = frame
    Instance.new("UICorner", scanBtn)

    local tpBtn = Instance.new("TextButton")
    tpBtn.Size = UDim2.new(.4,0,0,35)
    tpBtn.Position = UDim2.new(.55,0,1,-45)
    tpBtn.Text = "▶ Auto TP"
    tpBtn.Parent = frame
    Instance.new("UICorner", tpBtn)

    local function isCheckpoint(v)
        return v:IsA("Part") and v.Name == "Point" and v.Parent and tonumber(v.Parent.Name) ~= nil
    end

    local function scanAll()
        local found = {}

        local function recurse(parent)
            for _, v in ipairs(parent:GetChildren()) do
                if isCheckpoint(v) then
                    table.insert(found, v)
                end
                recurse(v)
            end
        end

        recurse(game.Workspace)
        return found
    end

    local function getSorted(list)
        table.sort(list, function(a,b)
            return tonumber(a.Parent.Name) < tonumber(b.Parent.Name)
        end)
        return list
    end

    local checkpoints = {}
    local running = false

    scanBtn.MouseButton1Click:Connect(function()
        checkpoints = getSorted(scanAll())
        info.Text = "Scan selesai!\n"..#checkpoints.." checkpoint ditemukan."
    end)

    tpBtn.MouseButton1Click:Connect(function()
        if #checkpoints == 0 then
            info.Text = "⚠️ Scan dulu!"
            return
        end

        running = not running

        task.spawn(function()
            for i, cp in ipairs(checkpoints) do
                if not running then break end

                local char = player.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")

                if cp and root then
                    info.Text = "🏁 Stage "..cp.Parent.Name
                    root.CFrame = CFrame.new(cp.Position + Vector3.new(0,3,0))
                    task.wait(0.7)
                end
            end

            running = false
            info.Text = "✅ Selesai!"
        end)
    end)
end
