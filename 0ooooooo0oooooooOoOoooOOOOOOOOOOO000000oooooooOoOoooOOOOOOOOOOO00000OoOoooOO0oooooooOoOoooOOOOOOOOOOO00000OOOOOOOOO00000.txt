---------------- SCAN ----------------

local function isCheckpoint(v)
    return v:IsA("Part")
        and v.Name == "Point"
        and v.Parent
        and tonumber(v.Parent.Name) ~= nil
end

local function scanAll()
    local found = {}

    local function recurse(parent)
        for _, v in ipairs(parent:GetChildren()) do
            if isCheckpoint(v) then
                table.insert(found, v)
            end
            recurse(v)
        end
    end

    recurse(game.Workspace)
    return found
end

local function getSorted(list)
    table.sort(list, function(a, b)
        return tonumber(a.Parent.Name) < tonumber(b.Parent.Name)
    end)
    return list
end

---------------- LOGIC ----------------

local checkpoints = {}

scanBtn.MouseButton1Click:Connect(function()
    checkpoints = getSorted(scanAll())

    info.Text = "Scan selesai!\n"..#checkpoints.." checkpoint ditemukan."

    tpBtn.BackgroundColor3 = #checkpoints > 0
        and Color3.fromRGB(30,150,60)
        or Color3.fromRGB(60,60,60)
end)

game.Workspace.DescendantAdded:Connect(function(v)
    if isCheckpoint(v) then
        for _, existing in ipairs(checkpoints) do
            if existing == v then return end
        end
        table.insert(checkpoints, v)
        checkpoints = getSorted(checkpoints)
    end
end)

---------------- AUTO TP ----------------

local running = false

tpBtn.MouseButton1Click:Connect(function()
    if #checkpoints == 0 then
        info.Text = "⚠️ Scan dulu!"
        return
    end

    running = not running

    tpBtn.Text = running and "⏹ Stop" or "▶ Auto TP"
    tpBtn.BackgroundColor3 = running
        and Color3.fromRGB(180,40,40)
        or Color3.fromRGB(30,150,60)

    if not running then return end

    task.spawn(function()
        for i, cp in ipairs(checkpoints) do
            if not running then break end

            local char = player.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")

            if cp and cp.Parent and root then
                info.Text = "🏁 Stage "..cp.Parent.Name.." ("..i.."/"..#checkpoints..")"
                root.CFrame = CFrame.new(cp.Position + Vector3.new(0,3,0))
                task.wait(0.7)
            end
        end

        if running then
            running = false
            tpBtn.Text = "▶ Auto TP"
            tpBtn.BackgroundColor3 = Color3.fromRGB(30,150,60)
            info.Text = "✅ Semua checkpoint selesai!"
        end
    end)
end)
